# Python 3.11+ required (onnxruntime floor). See README.md.

# Optimum 1.x keeps the ONNX exporters in-package and matches the
# WhisperOnnxConfig API that onnx_config.py subclasses. 2.x moved them to the
# separate `optimum-onnx` distribution; the scripts fall back to that import
# path, but this is the combination they were written against.
optimum[exporters]>=1.24,<2

# Pinned below 4.50: optimum 1.x tracks the pre-4.50 Whisper modelling code, and
# the export traces that module directly.

# Upper bound is not cosmetic: optimum 1.x's `exporters/onnx/model_patcher.py`
# imports `_attention_scale` from `torch.onnx.symbolic_opset14`, and torch
# removed it along with the rest of the legacy TorchScript ONNX exporter. With a
# newer torch, importing optimum's exporter fails outright.
torch>=2.2,<2.6

onnx>=1.17
onnxruntime>=1.20
# fp16 conversion in quantize.py
onnxconverter-common>=1.16
# onnxruntime >=1.28 imports onnx_ir from matmul_nbits_quantizer but does not
# declare it, so q4 quantisation fails with ModuleNotFoundError without this.
onnx-ir>=0.1

numpy>=1.26
